const unordered_map<string, function<bool(double, double)>> OPERATORS = {
    {">",  [](double a, double b) { return a > b;  }},
    {"<",  [](double a, double b) { return a < b;  }},
    {">=", [](double a, double b) { return a >= b; }},
    {"<=", [](double a, double b) { return a <= b; }},
    {"==", [](double a, double b) { return a == b; }},
    {"!=", [](double a, double b) { return a != b; }},
};

bool compare(double left, const string& op, double right) {
    auto it = OPERATORS.find(op);
    if (it == OPERATORS.end()) {
        return false;
    }
    return it->second(left, right);
}
